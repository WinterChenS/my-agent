# Task 10.6 气泡内图片与图片预览 — 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让「本轮产出的图片」在助手气泡**内**直接显示、在右侧「产物/来源文件」里都能预览，并让模型正文里按文件名引用的图片也能渲染。

**Architecture:** 归属问题在后端修（扫描路径补上本轮 `runId`，复用已存在的 `store(..., runId)` 重载）；展示问题在前端修（一个共享的图片判定 + 一个共享的 blob-URL hook + 一处 react-markdown `img` 覆盖）。不新增接口、不新增表字段、不改官方扩展。

**Tech Stack:** Java 21 / Spring Boot / MyBatis-Plus / JUnit5 + Mockito + AssertJ；React 18 + TypeScript + react-markdown + Vitest + Playwright。

## Global Constraints

```java
package com.myagent.artifact.service;

import java.util.Locale;

/**
 * 「扩展名 → MIME」的**唯一**实现。
 *
 * <p>此前交付路径与扫描路径各有一份，且已经不一致（前者认 webp/svg、后者不认）——
 * 结果是 Agent 写出的 svg/webp 图片在扫描路径下变成 application/octet-stream：
 * 既不按图片显示，也谈不上预览。这里合并成一份，两边都委托过来。
 */
public final class ArtifactMime {

    private ArtifactMime() {}

    public static String guess(String fileName) {
        String name = fileName == null ? "" : fileName.toLowerCase(Locale.ROOT);
        int dot = name.lastIndexOf('.');
        String ext = dot < 0 ? "" : name.substring(dot + 1);
        return switch (ext) {
            case "md", "markdown" -> "text/markdown";
            case "txt", "log" -> "text/plain";
            case "json", "jsonl" -> "application/json";
            case "csv" -> "text/csv";
            case "xml" -> "application/xml";
            case "yaml", "yml" -> "application/yaml";
            case "html", "htm" -> "text/html";
            case "css" -> "text/css";
            case "js", "mjs" -> "text/javascript";
            case "ts", "tsx", "py", "java", "sh", "sql", "properties", "ini", "conf", "toml" ->
                    "text/plain";
            case "png" -> "image/png";
            case "jpg", "jpeg" -> "image/jpeg";
            case "gif" -> "image/gif";
            case "webp" -> "image/webp";
            case "svg" -> "image/svg+xml";
            case "pdf" -> "application/pdf";
            case "zip" -> "application/zip";
            default -> "application/octet-stream";
        };
    }
}
```

新建 `frontend/src/components/chat/useArtifactBlobUrl.ts`：

```ts
import { useEffect, useState } from "react";
import { fetchArtifactBlobUrl, isImageArtifact, type ArtifactInfo } from "@/api/artifacts";

/**
 * 图片产物 → 可用于 `<img src>` 的 blob URL。
 *
 * <p>产物下载接口要带 `Authorization`，所以不能把 URL 直接塞给 `<img>`；统一走
 * `fetchArtifactBlobUrl`（fetch → blob URL）。**只有这一处**做这件事，四个地方复用：
 * 气泡内图片、产物缩略图、右侧抽屉、正文 Markdown 图片。卸载时自动 `revokeObjectURL`。
 */
export function useArtifactBlobUrl(item: ArtifactInfo | null | undefined) {
  const [url, setUrl] = useState<string | null>(null);
  const [failed, setFailed] = useState(false);
  const id = item?.id ?? null;
  const image = isImageArtifact(item);

  useEffect(() => {
    setUrl(null);
    setFailed(false);
    if (id == null || !image) {
      return;
    }
    let cancelled = false;
    let created: string | null = null;
    fetchArtifactBlobUrl(id)
      .then((next) => {
        created = next;
        if (cancelled) {
          URL.revokeObjectURL(next);
        } else {
          setUrl(next);
        }
      })
      .catch(() => {
        if (!cancelled) {
          setFailed(true);
        }
      });
    return () => {
      cancelled = true;
      if (created) {
        URL.revokeObjectURL(created);
      }
    };
  }, [id, image]);

  return { url, failed };
}
```

```ts
import { isImageArtifact, type ArtifactInfo } from "@/api/artifacts";

/**
 * 正文里 Markdown 图片的解析规则。
 *
 * <p>为什么按**文件名**匹配：官方 `deliver_artifact` 只回一句文字、不含 URL，模型无从得知
 * `/api/artifacts/{id}/download`，所以"模型写文件名、平台去产物里找"是唯一可行的约定；
 * 因此也**不新增** URL 协议、不改官方工具的返回值。
 */
export type ResolvedMarkdownImage =
  | { kind: "direct"; url: string }
  | { kind: "artifact"; item: ArtifactInfo };

export function resolveMarkdownImage(
  src: string,
  artifacts: ArtifactInfo[],
): ResolvedMarkdownImage | null {
  if (!src) {
    return null;
  }
  if (/^(data:|blob:|https?:)/i.test(src)) {
    return { kind: "direct", url: src };
  }
  const name = src.split(/[\\/]/).pop()?.toLowerCase() ?? "";
  if (!name) {
    return null;
  }
  const hit = artifacts.find((a) => isImageArtifact(a) && a.fileName.toLowerCase() === name);
  return hit ? { kind: "artifact", item: hit } : null;
}
```

/** 正文里的 Markdown 图片：直接渲染 / 按文件名找产物 / 给提示。 */
function MarkdownImage({
  src,
  alt,
  artifacts,
  onOpenArtifact,
}: {
  src: string;
  alt: string;
  artifacts: ArtifactInfo[];
  onOpenArtifact?: (item: ArtifactInfo) => void;
}) {
  const resolved = useMemo(() => resolveMarkdownImage(src, artifacts), [src, artifacts]);
  const artifactItem = resolved?.kind === "artifact" ? resolved.item : null;
  const { url, failed } = useArtifactBlobUrl(artifactItem);

  if (!resolved) {
    return (
      <span data-testid="markdown-image-missing" className="text-muted-foreground text-xs">
        图片未找到：{src}
      </span>
    );
  }
  if (resolved.kind === "direct") {
    return <img src={resolved.url} alt={alt} loading="lazy" className="max-h-64 rounded-lg border object-contain" />;
  }
  if (failed || !url) {
    return <span className="text-muted-foreground text-xs">{failed ? "图片加载失败" : "图片加载中…"}</span>;
  }
  return (
    <button type="button" title="点击查看大图" onClick={() => onOpenArtifact?.(resolved.item)} className="block">
      <img src={url} alt={alt || resolved.item.fileName} loading="lazy" className="max-h-64 rounded-lg border object-contain" />
    </button>
  );
}
